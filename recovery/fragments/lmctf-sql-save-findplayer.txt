
void Cmd_FindPlayer_f(edict_t* ent)
{

    char* err_msg = 0;
    char* playername;
    char* sql[MAX_MSGLEN];
    char* strbuff[MAX_MSGLEN];
    sqlite3* db;
    char* argstring;
    int* stat = STATS_ASSISTS;
    argstring = gi.args();

    if (!argstring || argstring[0] == 0)
    {
        playername = ent->client->pers.netname;
    }
    else
    {
        playername = argstring;
    }

    int rc = sqlite3_open(LMCTF_DATABASE, &db);

    sprintf(strbuff, "SELECT player_id FROM players WHERE player_name = \'%s\';", playername);
    strcpy(sql, strbuff);

    rc = sqlite3_exec(db, sql, callback, 0, &err_msg);

    if (rc != SQLITE_OK) {

        gi.dprintf("Failed to select data\n");
        gi.dprintf("SQL error: %s\n", err_msg);

        sqlite3_free(err_msg);
        sqlite3_close(db);
    }

    sqlite3_close(db);
}
